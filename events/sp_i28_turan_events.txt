namespace = sp_i28

# 1 — Post-Tanzimat imperial identity choice (one-shot).
sp_i28.1 = {
	type = country_event
	placement = root
	title = sp_i28.1.t
	desc = sp_i28.1.d
	flavor = sp_i28.1.f
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 3
	event_image = {
		video = "unspecific_politicians_arguing"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		sp_i28_identity_eligible = yes
	}

	immediate = {
		set_variable = sp_imperial_identity_chosen
	}

	option = {
		name = sp_i28.1.a
		default_option = yes
		set_variable = sp_ottomanism_selected
		custom_tooltip = sp_i28.1.a_tt
		ai_chance = {
			base = 50
		}
	}

	option = {
		name = sp_i28.1.b
		set_variable = sp_islamism_selected
		custom_tooltip = sp_i28.1.b_tt
		ai_chance = {
			base = 30
		}
	}

	option = {
		name = sp_i28.1.c
		set_variable = sp_turkism_selected
		custom_tooltip = sp_i28.1.c_tt
		sp_i28_try_add_turan_je = yes
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					country_rank >= rank_value:great_power
					has_technology_researched = pan-nationalism
					ig:ig_armed_forces ?= {
						is_powerful = yes
					}
				}
				add = 5
			}
		}
	}
}

# 10 — Stage 1 complete → Stage 2.
sp_i28.10 = {
	type = country_event
	placement = root
	title = sp_i28.10.t
	desc = sp_i28.10.d
	flavor = sp_i28.10.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
	event_image = {
		video = "europenorthamerica_before_the_battle"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		var:sp_turan_stage = 1
		NOT = { has_variable = sp_turan_stage_1_done }
		sp_i28_stage_1_complete = yes
	}

	immediate = {
		set_variable = sp_turan_stage_1_done
		set_variable = {
			name = sp_turan_stage
			value = 2
		}
		set_variable = {
			name = sp_turan_stage_goal
			value = 9
		}
		sp_i28_grant_stage_2_claims = yes
		sp_i28_recount_stage_progress = yes
	}

	option = {
		name = sp_i28.10.a
		default_option = yes
		add_modifier = {
			name = sp_i28_stage_complete_prestige
			months = 60
		}
	}
}
