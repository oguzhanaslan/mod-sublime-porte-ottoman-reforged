# 10 — Stage 1 complete → Stage 2.
sp_i31o.10 = {
	type = country_event
	placement = root
	title = sp_i31o.10.t
	desc = sp_i31o.10.d
	flavor = sp_i31o.10.f
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		var:sp_ottomanism_stage = 1
		NOT = { has_variable = sp_ottomanism_stage_1_done }
		sp_i31_ottomanism_stage_1_complete = yes
	}

	immediate = {
		set_variable = sp_ottomanism_stage_1_done
		set_variable = {
			name = sp_ottomanism_stage
			value = 2
		}
		set_variable = {
			name = sp_ottomanism_stage_goal
			value = 3
		}
		sp_i31_ottomanism_recount_progress = yes
	}

	option = {
		name = sp_i31o.10.a
		default_option = yes
		add_modifier = {
			name = sp_i31_stage_complete_prestige
			months = 60
		}
	}
}
