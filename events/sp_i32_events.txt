namespace = sp_i32

# Destination picker. Not fired from pulse.
sp_i32.11 = {
	type = country_event
	placement = root
	title = sp_i32.11.t
	desc = sp_i32.11.d
	flavor = sp_i32.11.f
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 14
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		has_journal_entry = sp_je_learning_from_europe
		NOT = { sp_i32_students_busy = yes }
	}

	immediate = {
		c:FRA ?= { save_scope_as = sp_i32_fra }
		c:GBR ?= { save_scope_as = sp_i32_gbr }
		c:AUS ?= { save_scope_as = sp_i32_aus }
		if = {
			limit = { exists = c:GER }
			c:GER = { save_scope_as = sp_i32_german }
		}
		else_if = {
			limit = { exists = c:PRU }
			c:PRU = { save_scope_as = sp_i32_german }
		}
	}

	option = {
		name = sp_i32.11.fra
		trigger = {
			gold_reserves >= sp_i32_students_upfront_standard
			custom_tooltip = {
				text = sp_i32_students_fra_diplo_tt
				sp_i32_students_can_use_country = { PARTNER = c:FRA }
			}
		}
		show_as_unavailable = {
			NOT = { sp_i32_students_fra_eligible = yes }
		}
		add_treasury = {
			value = sp_i32_students_pay_standard
		}
		je:sp_je_learning_from_europe = {
			add_modifier = {
				name = sp_i32_students_fra_act
				years = 6
			}
		}
		hidden_effect = {
			sp_i32_start_students_france = yes
		}
		ai_chance = { base = 40 }
	}

	option = {
		name = sp_i32.11.gbr
		trigger = {
			gold_reserves >= sp_i32_students_upfront_standard
			custom_tooltip = {
				text = sp_i32_students_gbr_diplo_tt
				sp_i32_students_can_use_country = { PARTNER = c:GBR }
			}
		}
		show_as_unavailable = {
			NOT = { sp_i32_students_gbr_eligible = yes }
		}
		add_treasury = {
			value = sp_i32_students_pay_standard
		}
		je:sp_je_learning_from_europe = {
			add_modifier = {
				name = sp_i32_students_gbr_act
				years = 6
			}
		}
		hidden_effect = {
			sp_i32_start_students_britain = yes
		}
		ai_chance = { base = 25 }
	}

	option = {
		name = sp_i32.11.ger
		trigger = {
			gold_reserves >= sp_i32_students_upfront_standard
			custom_tooltip = {
				text = sp_i32_students_german_diplo_tt
				sp_i32_students_german_host_ok = yes
			}
		}
		show_as_unavailable = {
			NOT = { sp_i32_students_german_eligible = yes }
		}
		add_treasury = {
			value = sp_i32_students_pay_standard
		}
		if = {
			limit = {
				sp_i32_mil_spread_occupied = yes
			}
			je:sp_je_learning_from_europe = {
				add_modifier = {
					name = sp_i32_students_pru_act_qual_only
					years = 6
				}
			}
		}
		else = {
			je:sp_je_learning_from_europe = {
				add_modifier = {
					name = sp_i32_students_pru_act_spread
					years = 6
				}
			}
		}
		hidden_effect = {
			sp_i32_start_students_german = yes
		}
		ai_chance = { base = 15 }
	}

	option = {
		name = sp_i32.11.aus
		trigger = {
			gold_reserves >= sp_i32_students_upfront_austria
			custom_tooltip = {
				text = sp_i32_students_aus_diplo_tt
				sp_i32_students_can_use_country = { PARTNER = c:AUS }
			}
		}
		show_as_unavailable = {
			NOT = { sp_i32_students_aus_eligible = yes }
		}
		add_treasury = {
			value = sp_i32_students_pay_austria
		}
		je:sp_je_learning_from_europe = {
			add_modifier = {
				name = sp_i32_students_aus_act
				years = 6
			}
		}
		hidden_effect = {
			sp_i32_start_students_austria = yes
		}
		ai_chance = { base = 20 }
	}

	option = {
		name = sp_i32.11.none
		default_option = yes
		ai_chance = { base = 0 }
	}
}

# Cancel confirmation.
sp_i32.30 = {
	type = country_event
	placement = root
	title = sp_i32.30.t
	desc = sp_i32.30.d
	flavor = sp_i32.30.f
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	duration = 7
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		has_variable = sp_i32_students_active
		NOT = { has_variable = sp_i32_students_completing }
	}

	immediate = {
		sp_i32_save_students_partner_scope = yes
	}

	option = {
		name = sp_i32.30.a
		default_option = yes
		custom_tooltip = sp_i32_students_cancel_cooldown_tt
		custom_tooltip = sp_i32_students_no_refund_tt
		if = {
			limit = {
				exists = scope:sp_i32_partner
			}
			change_relations = {
				country = scope:sp_i32_partner
				value = -5
			}
		}
		hidden_effect = {
			sp_i32_cancel_students_cleanup = yes
		}
		ai_chance = { base = 0 }
	}

	option = {
		name = sp_i32.30.b
		ai_chance = { base = 100 }
	}
}

# Students midpoint complication. Type rolled in immediate; programme continues.
sp_i32.50 = {
	type = country_event
	placement = root
	title = sp_i32.50.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = sp_i32_comp_overrun }
				desc = sp_i32.50.d_overrun
			}
			triggered_desc = {
				trigger = { has_variable = sp_i32_comp_delay }
				desc = sp_i32.50.d_delay
			}
			triggered_desc = {
				trigger = { has_variable = sp_i32_comp_remain }
				desc = sp_i32.50.d_remain
			}
			triggered_desc = {
				trigger = { has_variable = sp_i32_comp_radicals }
				desc = sp_i32.50.d_radicals
			}
			desc = sp_i32.50.d
		}
	}
	flavor = sp_i32.50.f
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	duration = 14
	event_image = {
		video = "unspecific_politicians_arguing"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		has_variable = sp_i32_students_active
	}

	immediate = {
		remove_variable = sp_i32_comp_overrun
		remove_variable = sp_i32_comp_delay
		remove_variable = sp_i32_comp_remain
		remove_variable = sp_i32_comp_radicals
		random_list = {
			25 = { set_variable = sp_i32_comp_overrun }
			25 = { set_variable = sp_i32_comp_delay }
			25 = { set_variable = sp_i32_comp_remain }
			25 = { set_variable = sp_i32_comp_radicals }
		}
		sp_i32_save_students_partner_scope = yes
	}

	option = {
		name = sp_i32.50.a
		default_option = yes
		if = {
			limit = { has_variable = sp_i32_comp_overrun }
			if = {
				limit = { has_variable = sp_i32_students_partner_aus }
				add_modifier = {
					name = sp_i32_students_overrun_aus
					years = 1
				}
			}
			else = {
				add_modifier = {
					name = sp_i32_students_overrun
					years = 1
				}
			}
		}
		if = {
			limit = { has_variable = sp_i32_comp_delay }
			custom_tooltip = sp_i32_students_extend_year_tt
			hidden_effect = {
				change_variable = {
					name = sp_i32_students_months_left
					add = 12
				}
			}
		}
		if = {
			limit = { has_variable = sp_i32_comp_remain }
			custom_tooltip = sp_i32_students_extend_year_tt
			hidden_effect = {
				change_variable = {
					name = sp_i32_students_months_left
					add = 12
				}
			}
			if = {
				limit = { exists = scope:sp_i32_partner }
				change_relations = {
					country = scope:sp_i32_partner
					value = 5
				}
			}
		}
		if = {
			limit = { has_variable = sp_i32_comp_radicals }
			add_modifier = {
				name = sp_i32_students_radicals
				years = 2
			}
		}
		hidden_effect = {
			remove_variable = sp_i32_comp_overrun
			remove_variable = sp_i32_comp_delay
			remove_variable = sp_i32_comp_remain
			remove_variable = sp_i32_comp_radicals
		}
	}
}
