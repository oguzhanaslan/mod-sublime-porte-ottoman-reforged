namespace = sp_i25

# I-25 Ottoman everyday flavor. Ten one-shot TUR country events. No JE, no character spawn,
# no laws, no tech, no buildings, no treasury, no Tanzimat points, no permanent modifiers.
# Cooldown is set only in immediate so a queued trigger_event cannot fail flavor_ready.

# 1 — The Men of the Porte, 1846–1860. No create_character (I-10A already has Âlî/Fuad).
sp_i25.1 = {
	type = country_event
	placement = root
	title = sp_i25.1.t
	desc = sp_i25.1.d
	flavor = sp_i25.1.f
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
	event_image = {
		video = "unspecific_politicians_arguing"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		sp_i25_can_fire_1 = yes
	}

	immediate = {
		set_variable = sp_i25_porte_men_fired
		sp_i25_set_flavor_cooldown = yes
	}

	option = {
		name = sp_i25.1.a
		default_option = yes
		add_modifier = {
			name = sp_i25_porte_new_men
			days = very_short_modifier_time
		}
		add_loyalists = {
			value = very_small_radicals
			pop_type = academics
		}
		ai_chance = {
			base = 40
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
				}
				add = 15
			}
		}
	}

	option = {
		name = sp_i25.1.b
		add_modifier = {
			name = sp_i25_porte_old_balance
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 60
			modifier = {
				trigger = {
					ig:ig_landowners ?= {
						is_in_government = yes
					}
				}
				add = 15
			}
		}
	}
}

	}

	immediate = {
		set_variable = sp_i25_petitions_fired
		sp_i25_set_flavor_cooldown = yes
		random_scope_state = {
			limit = {
				is_incorporated = yes
				OR = {
					turmoil >= 0.10

		}
	}
