# Wave B - Circassian centerpiece
sp_muhacir.2 = {
	type = country_event
	placement = root
	title = sp_muhacir.2.t
	desc = sp_muhacir.2.d
	flavor = sp_muhacir.2.f
	duration = 3
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		this = c:TUR
		NOT = { has_variable = sp_i12_wave_b_done }
		sp_i12_wave_b_can_fire = yes
	}

	immediate = {
		if = {
			limit = {
				sp_i12_vanilla_circassian_pipeline = yes
			}
			sp_i12_wave_b_set_vanilla_mode = yes
		}
		else = {
			sp_i12_wave_b_set_conservation_mode = yes
		}
	}

	option = {
		name = sp_muhacir.2.a
		sp_i12_wave_b_option_rumelia = yes
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					highest_secession_progress >= 0.30
				}
				add = -15
			}
		}
	}

	option = {
		name = sp_muhacir.2.b
		sp_i12_wave_b_option_anatolia = yes
		ai_chance = {
			base = 30
			modifier = {
				trigger = {
					highest_secession_progress >= 0.30
				}
				add = 20
			}
		}
	}

	option = {
		name = sp_muhacir.2.c
		default_option = yes
		sp_i12_wave_b_option_balanced = yes
		ai_chance = {
			base = 45
		}
	}

	option = {
		name = sp_muhacir.2.e
		sp_i12_wave_b_option_refuse = yes
		ai_chance = {
			base = 5
		}
	}
}
