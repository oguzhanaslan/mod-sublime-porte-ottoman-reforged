# 10 — Stage 1 → Stage 2 Arabia claims.
sp_i31i.10 = {
	type = country_event
	placement = root
	title = sp_i31i.10.t
	desc = sp_i31i.10.d
	flavor = sp_i31i.10.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		var:sp_islamism_stage = 1
		NOT = { has_variable = sp_islamism_stage_1_done }
		sp_i31_islamism_stage_1_complete = yes
	}

	immediate = {
		set_variable = sp_islamism_stage_1_done
		set_variable = {
			name = sp_islamism_stage
			value = 2
		}
		set_variable = {
			name = sp_islamism_stage_goal
			value = 4
		}
		sp_i31_grant_arabia_claims = yes
		sp_i31_apply_isolation_tier = yes
		sp_i31_islamism_recount_progress = yes
	}

	option = {
		name = sp_i31i.10.a
		default_option = yes
		add_modifier = {
			name = sp_i31_stage_complete_prestige
			months = 60
		}
	}
}

# 31 — The Caliph Calls (eligible Muslim country response).
sp_i31i.31 = {
	type = country_event
	placement = root
	title = sp_i31i.31.t
	desc = sp_i31i.31.d
	flavor = sp_i31i.31.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 14
	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		exists = c:TUR
		sp_i31_is_muslim_country = yes
		is_subject = no
		NOT = { THIS = c:TUR }
	}

	option = {
		name = sp_i31i.31.a
		default_option = yes
		change_relations = {
			country = c:TUR
			value = 20
		}
		c:TUR ?= {
			add_modifier = {
				name = sp_i31_caliph_support_prestige
				months = 24
			}
		}
		custom_tooltip = sp_i31i.31.a_tt
		ai_chance = {
			base = 40
			modifier = {
				trigger = {
					relations:c:TUR >= 20
				}
				add = 20
			}
			modifier = {
				trigger = {
					OR = {
						is_in_same_power_bloc = c:TUR
						any_scope_treaty = {
							binds = c:TUR
							any_scope_article = {
								OR = {
									has_type = alliance
									has_type = defensive_pact
								}
							}
						}
					}
				}
				add = 25
			}
		}
	}

	option = {
		name = sp_i31i.31.b
		custom_tooltip = sp_i31i.31.b_tt
		ai_chance = {
			base = 40
		}
	}

	option = {
		name = sp_i31i.31.c
		change_relations = {
			country = c:TUR
			value = -10
		}
		custom_tooltip = sp_i31i.31.c_tt
		ai_chance = {
			base = 20
			modifier = {
				trigger = {
					relations:c:TUR < 0
				}
				add = 20
			}
		}
	}
}
