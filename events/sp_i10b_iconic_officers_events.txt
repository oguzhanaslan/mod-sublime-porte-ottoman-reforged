namespace = sp_i10b

sp_i10b.1 = { # Fevzi Çakmak
	type = country_event
	placement = root
	gui_window = event_window_1char_tabloid
	left_icon = scope:sp_i10b_fevzi_scope

	title = sp_i10b.1.t
	desc = sp_i10b.1.d
	flavor = sp_i10b.1.f

	event_image = {
		texture = "gfx/event_pictures/sp_i10b_officer_event_scene.dds"
	}
	icon = "gfx/interface/icons/event_icons/sp_i10b_officer_event.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"
	duration = 3

	trigger = {
		this = c:TUR
		game_date >= 1850.1.1
		NOT = { has_variable = sp_i10b_fevzi_emerged }
		NOT = {
			any_scope_character = {
				has_template = sp_i10b_fevzi_cakmak_template
			}
		}
	}

	immediate = {
		create_character = {
			template = sp_i10b_fevzi_cakmak_template
			save_scope_as = sp_i10b_fevzi_scope
		}
		set_variable = sp_i10b_fevzi_emerged
	}

	option = {
		name = sp_i10b.1.a
		default_option = yes
		ai_chance = { base = 100 }
	}
}
