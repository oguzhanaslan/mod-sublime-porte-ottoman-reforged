namespace = sp_egyptian_question

sp_egyptian_question.1 = {
	type = country_event
	placement = root
	title = sp_egyptian_question.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { ruler = { has_template = tur_mahmut_osmanoglu_template } }
				desc = sp_egyptian_question.1.d.mahmud
			}
			triggered_desc = {
				trigger = { ruler = { has_template = tur_abdulmecid_osmanoglu_template } }
				desc = sp_egyptian_question.1.d.abdulmecid
			}
			desc = sp_egyptian_question.1.d
		}
	}
	flavor = sp_egyptian_question.1.f
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	duration = 3
	event_image = { video = "europenorthamerica_london_center" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	trigger = {
		c:TUR ?= this
		NOT = { has_variable = sp_egyptian_question_mediation_offered }
	}
	immediate = { set_variable = sp_egyptian_question_mediation_offered }

	option = {
		name = sp_egyptian_question.1.a
		default_option = yes
		custom_tooltip = sp_egyptian_question_historical_terms_tt
		c:EGY ?= { trigger_event = { id = sp_egyptian_question.2 popup = yes } }
		ai_chance = {
			base = 80
			modifier = { add = 20 sp_egyptian_question_coalition_leverage = yes }
			modifier = { add = 10 has_variable = sp_egyptian_question_battlefield_leverage }
		}
	}
	option = {
		name = sp_egyptian_question.1.b
		set_variable = sp_egyptian_question_pursuing_maximalist
		custom_tooltip = sp_egyptian_question_vanilla_sandbox_tt
		ai_chance = { base = 20 }
	}
}
