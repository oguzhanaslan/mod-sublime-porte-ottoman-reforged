namespace = sp_i21

# I-21 Kurdish provincial centralization. Four one-shot TUR events, 1836–1850.
# No JE, no character spawn, no forced revolt, no Kurdish subject.

# 1 — Eastern local authority, first monthly pulse on/after 1836.3.1.
# Monthly so it does not share TUR's autumn yearly pulse with I-23.1/.2.
sp_i21.1 = {
	type = country_event
	placement = scope:sp_i21_eastern_state_scope
	title = sp_i21.1.t
	desc = sp_i21.1.d
	flavor = sp_i21.1.f
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3
	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		sp_i21_arc_active = yes
		game_date >= 1836.3.1
		NOT = { has_variable = sp_i21_local_authority_fired }
		any_scope_state = {
			sp_i21_is_eastern_core_state = yes
		}
	}

	immediate = {
		set_variable = sp_i21_local_authority_fired
		random_scope_state = {
			limit = {
				sp_i21_is_eastern_core_state = yes
			}
			save_scope_as = sp_i21_eastern_state_scope
		}
	}

	option = {
		name = sp_i21.1.a
		default_option = yes
		every_scope_state = {
			limit = {
				sp_i21_is_eastern_core_state = yes
				is_incorporated = yes
			}
			add_modifier = {
				name = sp_i21_eastern_local_authority
			}
		}
		ai_chance = {
			base = 1
		}
	}
}

# 3 — Bedirhan: cooperation or command, 1843+. Event-only Bedirhan; hard SKIP if eastern stable.
sp_i21.3 = {
	type = country_event
	placement = scope:sp_i21_diyarbakir_scope
	title = sp_i21.3.t
	desc = sp_i21.3.d
	flavor = sp_i21.3.f
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		sp_i21_arc_active = yes
		game_date >= 1843.1.1
		has_variable = sp_i21_porte_reach_fired
		NOT = { has_variable = sp_i21_bedirhan_fired }
		NOT = { sp_i21_eastern_stable = yes }
		any_scope_state = {
			state_region = s:STATE_DIYARBAKIR
			is_incorporated = yes
		}
	}

	immediate = {
		set_variable = sp_i21_bedirhan_fired
		random_scope_state = {
			limit = {
				state_region = s:STATE_DIYARBAKIR
				is_incorporated = yes
			}
			save_scope_as = sp_i21_diyarbakir_scope
		}
	}

	option = {
		name = sp_i21.3.a
		default_option = yes
		set_variable = sp_i21_bedirhan_accommodated
		add_modifier = {
			name = sp_i21_local_accommodation
			days = very_short_modifier_time
		}
		if = {
			limit = {
				exists = scope:sp_i21_diyarbakir_scope
			}
			scope:sp_i21_diyarbakir_scope = {
				add_loyalists_in_state = {
					value = very_small_radicals
					culture = cu:kurdish
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					ig:ig_landowners ?= {
						is_in_government = yes
					}
				}
				add = 5
			}
		}
	}

	option = {
		name = sp_i21.3.b
		set_variable = sp_i21_bedirhan_enforced
		if = {
			limit = {
				exists = scope:sp_i21_diyarbakir_scope
			}
			scope:sp_i21_diyarbakir_scope = {
				add_radicals_in_state = {
					value = very_small_radicals
					culture = cu:kurdish
				}
			}
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
					sp_i21_centralization_pressure = yes
				}
				add = 5
			}
		}
	}
}

# 4 — Direct provincial administration, 1847+. Clears the eastern situation and grants a short provincial recovery.
sp_i21.4 = {
	type = country_event
	placement = scope:sp_i21_diyarbakir_scope
	title = sp_i21.4.t
	desc = sp_i21.4.d
	flavor = sp_i21.4.f
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3
	event_image = {
		video = "ep1_ca_mountain_pass_civilian"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		sp_i21_arc_active = yes
		game_date >= 1847.1.1
		has_variable = sp_i21_porte_reach_fired
		NOT = { has_variable = sp_i21_direct_admin_fired }
		NOT = { has_variable = sp_i21_arc_closed }
		OR = {
			AND = {
				has_variable = sp_i21_bedirhan_enforced
				sp_i21_has_eastern_admin_capacity = yes
			}
			AND = {
				has_variable = sp_i21_bedirhan_accommodated
				sp_i21_centralization_pressure = yes
				sp_i21_has_eastern_admin_capacity = yes
			}
			AND = {
				sp_i21_eastern_stable = yes
				NOT = { has_variable = sp_i21_bedirhan_fired }
			}
		}
	}

	immediate = {
		set_variable = sp_i21_direct_admin_fired
		set_variable = sp_i21_arc_closed
		random_scope_state = {
			limit = {
				state_region = s:STATE_DIYARBAKIR
				is_incorporated = yes
			}
			save_scope_as = sp_i21_diyarbakir_scope
		}
	}

	option = {
		name = sp_i21.4.a
		default_option = yes
		every_scope_state = {
			limit = {
				has_modifier = sp_i21_eastern_local_authority
			}
			remove_modifier = sp_i21_eastern_local_authority
		}
		every_scope_state = {
			limit = {
				has_modifier = sp_i21_porte_administrative_reach
			}
			remove_modifier = sp_i21_porte_administrative_reach
		}
		every_scope_state = {
			limit = {
				sp_i21_is_eastern_core_state = yes
				is_incorporated = yes
			}
			add_modifier = {
				name = sp_i21_direct_provincial_admin
				days = normal_modifier_time
			}
		}
		ai_chance = {
			base = 1
		}
	}
}
