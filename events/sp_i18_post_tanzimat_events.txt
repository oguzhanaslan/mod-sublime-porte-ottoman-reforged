# I-18 post-Tanzimat political flavor. Five one-shot TUR events. No JE, no laws, no movements.

# 1 — Post-Tanzimat debate, 1852–1864.
sp_i18.1 = {
	type = country_event
	placement = root
	title = sp_i18.1.t
	desc = sp_i18.1.d
	flavor = sp_i18.1.f

	duration = 3

		video = "unspecific_politicians_arguing"

	trigger = {
		sp_i18_phase_i_eligible = yes
	}

	immediate = {
		set_variable = sp_i18_debate_fired
	}

	option = {
		name = sp_i18.1.a
		default_option = yes
		add_modifier = {
			name = sp_i18_reform_debate

		}

				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}

				add = 20
			}
		}
	}

	option = {
		name = sp_i18.1.b
		add_modifier = {
			name = sp_i18_palace_prerogative
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 60
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
				}
				add = -20
			}
		}
	}
}

# 2 — Young Ottomans circle/press, 1865+. Does not create Nâmık or Midhat.
sp_i18.2 = {
	type = country_event
	placement = root
	title = sp_i18.2.t
	desc = sp_i18.2.d
	flavor = sp_i18.2.f
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3
	event_image = {
		video = "asia_politician_parliament_motion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		has_law = law_type:law_monarchy
		game_date >= 1865.1.1
		NOT = { has_variable = sp_i18_young_ottomans_fired }
		OR = {
			has_variable = sp_i18_debate_fired
			sp_i18_has_domestic_tanzimat_reform = yes
			has_variable = sick_man_complete
			has_variable = failed_sick_man_main
			has_variable = failed_sick_man_separatism
		}
		OR = {
			ig:ig_intelligentsia ?= {
				ig_counts_as_marginal = no
			}
			any_political_movement = {
				is_political_movement_type = movement_liberal
			}
			any_political_movement = {
				is_political_movement_type = movement_radical
			}
			has_law_or_variant = law_type:law_censorship
		}
	}

	immediate = {
		set_variable = sp_i18_young_ottomans_fired
	}

	option = {
		name = sp_i18.2.a
		default_option = yes
		# pop_type = academics is not attested on country-scope add_radicals.
		# I-13 / dominican country-event pattern: culture-filtered radicals.
		add_radicals = {
			value = very_small_radicals
			culture = cu:turkish
		}
		add_modifier = {
			name = sp_i18_young_ottoman_press
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					has_law_or_variant = law_type:law_censorship
				}
				add = 5
			}
		}
	}

	option = {
		name = sp_i18.2.b
		add_modifier = {
			name = sp_i18_young_ottoman_warning
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					ig:ig_landowners ?= {
						is_in_government = yes
					}
				}
				add = 5
			}
		}
	}
}

# 3 — Early constitutional acknowledgement, 1852–1875. Not Kanun-i Esasi. NO EFFECT.
sp_i18.3 = {
	type = country_event
	placement = root
	title = sp_i18.3.t
	desc = sp_i18.3.d
	flavor = sp_i18.3.f
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 3
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		has_law = law_type:law_monarchy
		game_date >= 1852.1.1
		game_date < 1876.1.1
		NOT = { has_variable = sp_i18_early_constitution_fired }
		OR = {
			is_enacting_law = law_type:law_landed_voting
			is_enacting_law = law_type:law_wealth_voting
			is_enacting_law = law_type:law_census_voting
			is_enacting_law = law_type:law_universal_suffrage
		}
	}

	immediate = {
		set_variable = sp_i18_early_constitution_fired
		set_variable = sp_i18_constitution_seen
	}

	option = {
		name = sp_i18.3.a
		default_option = yes
		ai_chance = {
			base = 1
		}
	}
}
