namespace = sp_i18

# I-18 post-Tanzimat political flavor. Six one-shot TUR events.
# Relative sequence after Tanzimat closes — no calendar year gates.

# 1 — Post-Tanzimat debate. After Tanzimat JE closes.
sp_i18.1 = {
	type = country_event
	placement = root
	title = sp_i18.1.t
	desc = sp_i18.1.d
	flavor = sp_i18.1.f
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
	event_image = {
		video = "unspecific_politicians_arguing"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		sp_i18_phase_i_eligible = yes
	}

	immediate = {
		set_variable = sp_i18_debate_fired
		set_variable = {
			name = sp_i18_yo_wait
			days = 365
		}
	}

	option = {
		name = sp_i18.1.a
		default_option = yes
		add_modifier = {
			name = sp_i18_reform_debate
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 40
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
				}
				add = 20
			}
		}
	}

	option = {
		name = sp_i18.1.b
		add_modifier = {
			name = sp_i18_palace_prerogative
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 60
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
				}
				add = -20
			}
		}
	}
}

# 2 — Young Ottomans circle/press. ~1 year after debate. Does not create Nâmık or Midhat.
sp_i18.2 = {
	type = country_event
	placement = root
	title = sp_i18.2.t
	desc = sp_i18.2.d
	flavor = sp_i18.2.f
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3
	event_image = {
		video = "asia_politician_parliament_motion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		has_law = law_type:law_monarchy
		has_variable = sp_i18_debate_fired
		NOT = { has_variable = sp_i18_yo_wait }
		NOT = { has_variable = sp_i18_young_ottomans_fired }
	}

	immediate = {
		set_variable = sp_i18_young_ottomans_fired
		set_variable = {
			name = sp_i18_palace_wait
			days = 365
		}
	}

	option = {
		name = sp_i18.2.a
		default_option = yes
		# pop_type = academics is not attested on country-scope add_radicals.
		# I-13 / dominican country-event pattern: culture-filtered radicals.
		add_radicals = {
			value = very_small_radicals
			culture = cu:turkish
		}
		add_modifier = {
			name = sp_i18_young_ottoman_press
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					has_law_or_variant = law_type:law_censorship
				}
				add = 5
			}
		}
	}

	option = {
		name = sp_i18.2.b
		add_modifier = {
			name = sp_i18_young_ottoman_warning
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 50
			modifier = {
				trigger = {
					ig:ig_landowners ?= {
						is_in_government = yes
					}
				}
				add = 5
			}
		}
	}
}

# 3 — Early constitutional acknowledgement on franchise law-pass. Not Kanun-i Esasi. NO EFFECT.
sp_i18.3 = {
	type = country_event
	placement = root
	title = sp_i18.3.t
	desc = sp_i18.3.d
	flavor = sp_i18.3.f
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	duration = 3
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		has_law = law_type:law_monarchy
		NOT = { has_variable = sp_i18_early_constitution_fired }
		OR = {
			is_enacting_law = law_type:law_landed_voting
			is_enacting_law = law_type:law_wealth_voting
			is_enacting_law = law_type:law_census_voting
			is_enacting_law = law_type:law_universal_suffrage
		}
	}

	immediate = {
		set_variable = sp_i18_early_constitution_fired
		set_variable = sp_i18_constitution_seen
	}

	option = {
		name = sp_i18.3.a
		default_option = yes
		ai_chance = {
			base = 1
		}
	}
}
