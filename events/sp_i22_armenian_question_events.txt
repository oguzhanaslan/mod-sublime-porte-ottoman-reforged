namespace = sp_i22

# I-22 Armenian Question. Two one-shot TUR events. No JE opener, no gameplay effects.

# 1 — Armenian National Constitution, 1863+. Communal administration acknowledgement only.
sp_i22.1 = {
	type = country_event
	placement = root
	title = sp_i22.1.t
	desc = sp_i22.1.d
	flavor = sp_i22.1.f
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		game_date >= 1863.1.1
		game_date < 1878.1.1
		NOT = { has_variable = sp_i22_communal_constitution_fired }
		NOT = { sp_i22_communal_constitution_skip = yes }
	}

	immediate = {
		set_variable = sp_i22_communal_constitution_fired
	}

	option = {
		name = sp_i22.1.a
		default_option = yes
		ai_chance = {
			base = 1
		}
	}
}
