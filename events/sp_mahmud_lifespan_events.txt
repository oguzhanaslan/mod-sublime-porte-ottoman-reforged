namespace = sp_mahmud

# Scripted 1849 death. Vanilla Alfonso XII / Sweden Oscar pattern: date gate + kill on the option.
# Abdülmecid is already heir; vanilla on_character_death creates Abdülaziz when Abdülmecid later dies.

sp_mahmud.1 = {
	type = country_event
	placement = root
	title = sp_mahmud.1.t
	desc = sp_mahmud.1.d
	flavor = sp_mahmud.1.f

	gui_window = event_window_1char_tabloid
	left_icon = scope:sp_mecid_scope

	icon = "gfx/interface/icons/law_icons/monarchy.dds"
	duration = 3
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		c:TUR ?= this
		game_date >= 1849.1.1
		NOT = { has_variable = sp_mahmud_death_fired }
		ruler ?= {
			has_template = tur_mahmut_osmanoglu_template
		}
		heir ?= {
			has_template = tur_abdulmecid_osmanoglu_template
		}
	}

	immediate = {
		set_variable = sp_mahmud_death_fired
		ruler = {
			save_scope_as = sp_mahmud_scope
		}
		heir = {
			save_scope_as = sp_mecid_scope
		}
	}

	option = {
		name = sp_mahmud.1.a
		default_option = yes
		scope:sp_mahmud_scope = {
			kill_character = yes
		}
		ai_chance = {
			base = 1
		}
	}
}
