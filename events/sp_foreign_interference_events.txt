namespace = sp_fi

# I-33 Foreign interference / Şark Meselesi. Russia-only. Recurring flavor +
# pressure, not a war/rebellion generator. Shared cooldown is set in immediate.

# 1 — Russian note to the Porte. Orthodox question from 1836.
sp_fi.1 = {
	type = country_event
	placement = root
	title = sp_fi.1.t
	desc = sp_fi.1.d
	flavor = sp_fi.1.f
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		sp_fi_can_fire_note = yes
	}

	immediate = {
		sp_fi_set_cooldown = yes
		c:RUS ?= {
			save_scope_as = sp_fi_rus
		}
	}

	option = {
		name = sp_fi.1.a
		default_option = yes
		change_relations = {
			country = c:RUS
			value = 10
		}
		add_loyalists = {
			value = very_small_radicals
			religion = rel:orthodox
		}
		add_modifier = {
			name = sp_fi_note_concession
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
				}
				add = 10
			}
		}
	}

	option = {
		name = sp_fi.1.b
		change_relations = {
			country = c:RUS
			value = -15
		}
		add_radicals = {
			value = very_small_radicals
			religion = rel:orthodox
		}
		add_modifier = {
			name = sp_fi_note_sovereignty
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 45
			modifier = {
				trigger = {
					ig:ig_landowners ?= {
						is_in_government = yes
					}
				}
				add = 10
			}
			modifier = {
				trigger = {
					ig:ig_devout ?= {
						is_in_government = yes
					}
				}
				add = 10
			}
		}
	}

	option = {
		name = sp_fi.1.c
		trigger = {
			gold_reserves >= sp_fi_reform_cost
		}
		add_treasury = {
			value = sp_fi_reform_cost
			multiply = -1
		}
		change_relations = {
			country = c:RUS
			value = -5
		}
		add_loyalists = {
			value = very_small_radicals
			religion = rel:orthodox
		}
		add_modifier = {
			name = sp_fi_note_reform
			days = very_short_modifier_time
		}
		ai_chance = {
			base = 30
			modifier = {
				trigger = {
					ig:ig_intelligentsia ?= {
						is_in_government = yes
					}
				}
				add = 15
			}
		}
	}
}
