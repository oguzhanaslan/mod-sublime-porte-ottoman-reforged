# I-32 Wave A+B+C+D — Students (0), Military (1), Technical (2), Education (3), Industrial (4), Naval (5).
# Loc: keep desc / second_desc short — CMF paints both on the bar face (~504px).

sp_i32_spb_students = {
	name = "sp_i32_spb_students"
	desc = "sp_i32_spb_students_desc"
	second_desc = "sp_i32_spb_students_second"

	default = yes

	start_value = 0
	min_value = 0
	max_value = 100

	# Presentation only. Authoritative timer is sp_i32_students_months_left.
	monthly_progress = {
		if = {
			limit = {
				has_variable = sp_i32_students_active
				NOT = { has_variable = sp_i32_students_completing }
			}
			add = {
				desc = "sp_i32_spb_students_tick"
				value = {
					value = 100
					divide = var:sp_i32_students_duration_months
				}
			}
		}
	}
}
