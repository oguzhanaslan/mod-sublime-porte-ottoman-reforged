# I-30 Phase 2 — Campaign Tree node select + state visibility scripted GUIs.
# Selection writes UI-only sp_ct_selected_node (flag). No gameplay progression.

### Selectors

sp_ct_select_tanzimat = {
	scope = country
	is_shown = { c:TUR ?= this }
	is_valid = { c:TUR ?= this }
	effect = {
		set_variable = {
			name = sp_ct_selected_node
			value = flag:sp_ct_node_tanzimat
		}
	}
}
