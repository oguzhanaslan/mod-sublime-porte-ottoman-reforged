# I-30 native tooltip pass — ExecuteTooltip sguis (effect is NOT run; renders as tooltip only).
# Pattern: vanilla je_india_railway_*_sgui + SCOPE.sStateRegion GetName.
# Visibility icons stay on sp_ct_vis_ctrl_*; these sguis are hover-only.
#
# Maintainability note: each territory needs its own STATE_* key in script — Clausewitz
# cannot parameterize state region keys inside one shared sgui. Keep one thin copy per
# state; do not invent a “generic” territory sgui without a verified vanilla pattern.
#
# Sections below:
#   1) Territory control (sp_ct_tt_ctrl_*) — shared loc sp_ct_terr_ctrl_held/_missing
#   2) Requirement rows (sp_ct_tt_req_*) — shared loc sp_ct_req_ctrl_held/_missing + label keys

### Shared entry lines (loc): sp_ct_terr_ctrl_held / sp_ct_terr_ctrl_missing

sp_ct_tt_ctrl_azerbaijan = {
	scope = country
	effect = {
		s:STATE_AZERBAIJAN = { save_temporary_scope_as = sp_ct_state_region }
		if = {
			limit = { country_or_subject_owns_entire_state_region = STATE_AZERBAIJAN }
			custom_tooltip_no_bullet = "sp_ct_terr_ctrl_held"
		}
		else = {
			custom_tooltip_no_bullet = "sp_ct_terr_ctrl_missing"
		}
	}
}

### Requirement ExecuteTooltip sguis (if/limit + loc lines; nested triggers invalid in sgui effect)

sp_ct_tt_req_pan_nationalism = {
	scope = country
	effect = {
		if = {
			limit = { has_technology_researched = pan-nationalism }
			custom_tooltip_no_bullet = "sp_ct_req_ctrl_held"
		}
		else = {
			custom_tooltip_no_bullet = "sp_ct_req_ctrl_missing"
		}
		custom_tooltip_no_bullet = "sp_ct_req_pan_nationalism"
	}
}
