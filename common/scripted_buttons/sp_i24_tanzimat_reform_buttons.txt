# I-24 one-shot Tanzimat reform buttons. Independent of Gates of the Bosphorus.
# Attached only to the additive companion JE sp_je_tanzimat_reforms.

sp_i24_abolish_cizye_button = {
	name = sp_i24_abolish_cizye_button
	desc = sp_i24_abolish_cizye_button_desc

	visible = {
		NOT = { sp_i24_cizye_done = yes }
	}

	possible = {
		custom_tooltip = {
			text = sp_i24_legitimacy_60_tt
			sp_i24_reform_legitimacy = yes
		}
		custom_tooltip = {
			text = sp_i24_devout_out_of_government_tt
			sp_i24_devout_resistance_weak = yes
		}
		hidden_trigger = {
			NOT = { sp_i24_cizye_done = yes }
		}
	}

	ai_chance = {
		value = 8
		if = {
			limit = {
				sp_i24_ai_reform_pause = yes
			}
			add = -1000
		}
	}

	effect = {
		set_variable = sp_i24_cizye_abolished
		add_modifier = {
			name = sp_i24_cizye_abolished_mod
			days = normal_modifier_time
		}
		sp_i24_add_cizye_loyalists_effect = yes
		custom_tooltip = sp_i24_no_tanzimat_point_tt
	}
}
