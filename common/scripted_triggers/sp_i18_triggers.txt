# I-18 post-Tanzimat political flavor. Reads I-05 vars; does not set them.
# sick_man_complete is not Tanzimat success (I-06 sets it on success and failure).

# At least one domestic Tanzimat objective completed. Syria/Egypt reconquest is not enough.
sp_i18_has_domestic_tanzimat_reform = {
	OR = {
		has_variable = completed_sick_man_bureaucracy_je
		has_variable = completed_sick_man_education_je
		has_variable = completed_sick_man_army_je
		has_variable = completed_sick_man_economy_je
		has_variable = completed_separatism_sick_man_je
	}
}

# Left at least one 1836 political/institutional baseline, or already has a franchise.
sp_i18_has_minimum_political_modernization = {
	OR = {
		NOT = { has_law_or_variant = law_type:law_hereditary_bureaucrats }
		NOT = { has_law_or_variant = law_type:law_peasant_levies }
		NOT = { has_law_or_variant = law_type:law_traditionalism }
		NOT = { has_law_or_variant = law_type:law_no_schools }
		NOT = { has_law = law_type:law_autocracy }
		sp_i18_has_constitutional_government = yes
	}
}

# Monarchy plus a voting franchise. Oligarchy and autocracy are not constitutional.
sp_i18_has_constitutional_government = {
	has_law = law_type:law_monarchy
	OR = {
		has_law_or_variant = law_type:law_landed_voting
		has_law_or_variant = law_type:law_wealth_voting
		has_law_or_variant = law_type:law_census_voting
		has_law_or_variant = law_type:law_universal_suffrage
	}
}

# Phase I: after Tanzimat resolves. Calendar years do not gate this beat.
sp_i18_phase_i_eligible = {
	c:TUR ?= this
	has_law = law_type:law_monarchy
	has_variable = sick_man_complete
	sp_i18_has_domestic_tanzimat_reform = yes
	sp_i18_has_minimum_political_modernization = yes
	NOT = { has_variable = sp_i18_debate_fired }
}

# Authoritative I-18 close. Same OR as sp_je_post_tanzimat_era complete:
# Kanun-i Esasi, Hamidian restoration, or palace path (no franchise after YO wait).
# No calendar floor.
sp_i18_arc_complete = {
	OR = {
		has_variable = sp_i18_kanun_i_esasi_fired
		has_variable = sp_i18_hamidian_fired
		has_variable = sp_i18_palace_path_fired
	}
}
