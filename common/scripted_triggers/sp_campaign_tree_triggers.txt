# I-30 Campaign Tree — read-only node state triggers.
# No progression writers. Priority: COMPLETED > ACTIVE > AVAILABLE > LOCKED.

### Shared gates

sp_ct_is_tur = {
	c:TUR ?= this
}

sp_ct_post_tanzimat_gate = {
	sp_i28_post_tanzimat = yes
}

sp_ct_post_tanzimat_arc_complete = {
	OR = {
		has_variable = sp_i18_hamidian_fired
		AND = {
			has_variable = sp_i18_kanun_i_esasi_fired
			game_date >= 1886.1.1
		}
	}
}

sp_ct_post_tanzimat_arc_active = {
	OR = {
		has_journal_entry = sp_je_post_tanzimat_era
		has_variable = sp_i18_debate_fired
		has_variable = sp_i18_young_ottomans_fired
		has_variable = sp_i18_early_constitution_fired
		has_variable = sp_i18_constitution_seen
		has_variable = sp_i18_kanun_i_esasi_fired
	}
	NOT = { sp_ct_post_tanzimat_arc_complete = yes }
}

			NOT = { sp_ct_tanzimat_available = yes }
		}
	}
}
