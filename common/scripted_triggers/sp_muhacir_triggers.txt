# Country scope (call as c:TUR).
sp_i12_has_devastated_held_rumelian_core = {
	any_scope_state = {
		devastation >= 15
		OR = {
			state_region = s:STATE_BULGARIA
			state_region = s:STATE_NORTHERN_THRACE
			state_region = s:STATE_WESTERN_THRACE
			state_region = s:STATE_MACEDONIA
			state_region = s:STATE_KOSOVO
			state_region = s:STATE_SKOPIA
			state_region = s:STATE_DOBRUDJA
			state_region = s:STATE_EASTERN_SERBIA
		}
	}
}

# Country scope (call as c:TUR).
sp_i12_wave_c_crisis_signal = {
	OR = {
		sp_i12_lost_rumelian_core = yes
		AND = {
			has_war_with = c:RUS
			sp_i12_has_devastated_held_rumelian_core = yes
		}
	}
}

# Country scope: vanilla Circassian pipeline proof (proposal 2.1).
sp_i12_vanilla_circassian_pipeline = {
	OR = {
		c:RUS ?= {
			has_journal_entry = je_circassian_expulsions
		}
		AND = {
			game_date >= 1863.1.1
			game_date < 1868.1.1
			c:RUS ?= {
				has_variable = milyutin_memorandum_accepted
			}
			NOT = { exists = c:CIR }
		}
		AND = {
			game_date >= 1863.1.1
			game_date < 1868.1.1
			has_variable = refused_circassian_refugees
		}
	}
}

# Country scope: Wave B conservation door.
sp_i12_wave_b_conservation_door = {
	NOT = { sp_i12_vanilla_circassian_pipeline = yes }
	game_date >= 1864.1.1
	game_date < 1868.1.1
	NOT = { exists = c:CIR }
	c:RUS ?= {
		owns_entire_state_region = STATE_KUBAN
	}
}

# Country scope: Wave B can fire.
sp_i12_wave_b_can_fire = {
	OR = {
		sp_i12_vanilla_circassian_pipeline = yes
		sp_i12_wave_b_conservation_door = yes
	}
}
