# I-32 Wave A+B+C+D — Learning from Europe. TUR-only. No Tanzimat gate.

sp_i32_is_tur = {
	c:TUR ?= this
}

sp_i32_has_staffed_university = {
	any_scope_building = {
		is_building_type = building_university
		occupancy >= 0.50
		level >= 1
	}
}

# Technical unlock: any university building at level ≥ 1 (occupancy not required).
sp_i32_has_university_building = {
	any_scope_building = {
		is_building_type = building_university
		level >= 1
	}
}

# Live incoming military_assistance. Not a year gate. Source is never TUR.
sp_i32_has_incoming_military_assistance = {
	any_scope_treaty = {
		any_scope_article = {
			has_type = military_assistance
			target_country = root
			NOT = { source_country = root }
		}
	}
}

sp_i32_has_incoming_ma_from_german_sphere = {
	OR = {
		AND = {
			exists = c:PRU
			any_scope_treaty = {
				binds = c:PRU
				any_scope_article = {
					has_type = military_assistance
					source_country = c:PRU
					target_country = root
				}
			}
		}
		AND = {
			exists = c:GER
			any_scope_treaty = {
				binds = c:GER
				any_scope_article = {
					has_type = military_assistance
					source_country = c:GER
					target_country = root
				}
			}
		}
	}
}

# MIL_SPREAD occupied: incoming MA, I-32 land with mil spread (not GBR QUAL-only),
# or naval mission using the spread variant (not QUAL-only).
sp_i32_mil_spread_occupied = {
	OR = {
		sp_i32_has_incoming_military_assistance = yes
		AND = {
			has_variable = sp_i32_military_active
			NOT = { has_variable = sp_i32_military_partner_gbr }
		}
		AND = {
			has_variable = sp_i32_naval_active
			NOT = { has_variable = sp_i32_naval_qual_only }
		}
	}
}

# Military family: MA OR I-32 land invite — OR so they never double-count.
sp_i32_military_family_occupied = {
	OR = {
		sp_i32_has_incoming_military_assistance = yes
		has_variable = sp_i32_military_active
	}
}

# I-07 Egypt / Eastern Question window (TUR country). Blocks French land invite only.
# Matches proposal + I-16B crisis signals (egyptian_crisis / Syria / Egypt JEs).
sp_i32_egypt_crisis_window = {
	NOT = { has_variable = sp_egyptian_question_resolved }
	OR = {
		has_variable = egyptian_crisis
		has_journal_entry = je_sick_man_syria
		has_journal_entry = je_sick_man_egypt
		has_journal_entry = sp_egyptian_question
	}
}

	}

# Hostile partner: war, Cold or worse, or antagonistic/belligerent/domineering attitude toward TUR.
sp_i32_students_can_use_country = {
	exists = $PARTNER$
	NOT = { has_war_with = $PARTNER$ }
	$PARTNER$ = {
		relations:root > -20
		NOR = {
			has_attitude = {
				who = root
				attitude = antagonistic
			}
			has_attitude = {
				who = root
				attitude = belligerent
			}
			has_attitude = {
				who = root
				attitude = domineering
			}
		}
	}
}

# Wave B partner gate: relations ≥ 0 (stricter than Students > −20).
sp_i32_partner_ok_neutral = {
	exists = $PARTNER$
	NOT = { has_war_with = $PARTNER$ }
	$PARTNER$ = {
		relations:root >= 0
		NOR = {
			has_attitude = {
				who = root
				attitude = antagonistic
			}
			has_attitude = {
				who = root
				attitude = belligerent
			}
			has_attitude = {
				who = root
				attitude = domineering
			}
		}
	}
}

# Developing: any two signals. Wave A stores no expertise mana.
sp_i32_expertise_developing = {
	calc_true_if = {
		amount >= 2
		institution_investment_level = {
			institution = institution_schools
			value >= 2
		}
		sp_i32_has_staffed_university = yes
		literacy_rate >= 0.22
		has_technology_researched = napoleonic_warfare
		has_technology_researched = mechanical_tools
	}
}

sp_i32_has_two_families_completed = {
	calc_true_if = {
		amount >= 2
		has_variable = sp_i32_students_ever_completed
		has_variable = sp_i32_military_ever_completed
		has_variable = sp_i32_technical_ever_completed
		has_variable = sp_i32_education_ever_completed
		has_variable = sp_i32_industrial_ever_completed
		has_variable = sp_i32_naval_ever_completed
	}
}

sp_i32_expertise_established = {
	sp_i32_expertise_developing = yes
	calc_true_if = {
		amount >= 2
		any_scope_building = {
			is_building_type = building_university
			occupancy >= 0.80
			level >= 2
		}
		literacy_rate >= 0.30
		AND = {
			has_variable = sp_i32_bur_surplus_months
			var:sp_i32_bur_surplus_months >= 24
		}
		has_technology_researched = general_staff
		sp_i32_has_two_families_completed = yes
	}
}

sp_i32_capacity_is_4 = {
	sp_i32_expertise_established = yes
}

sp_i32_capacity_is_3 = {
	NOT = { sp_i32_capacity_is_4 = yes }
	OR = {
		AND = {
			has_variable = sp_i32_bur_surplus_months
			var:sp_i32_bur_surplus_months >= 12
		}
		sp_i32_has_staffed_university = yes
	}
}

sp_i32_capacity_is_2 = {
	NOR = {
		sp_i32_capacity_is_3 = yes
		sp_i32_capacity_is_4 = yes
	}
}

			has_attitude = {
				who = root
				attitude = domineering
			}
		}
	}
	OR = {
		has_variable = sp_i32_primary_partner_gbr
		c:GBR = {
			relations:root >= 20
		}
	}
}
