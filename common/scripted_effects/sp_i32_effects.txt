sp_i32_init_capacity_cache = {
	if = {
		limit = {
			NOT = { has_variable = sp_i32_bur_surplus_months }
		}
		set_variable = {
			name = sp_i32_bur_surplus_months
			value = 0
		}
	}
}

sp_i32_tick_capacity_cache = {
	sp_i32_init_capacity_cache = yes
	if = {
		limit = {
			bureaucracy > 0
		}
		change_variable = {
			name = sp_i32_bur_surplus_months
			add = 1
		}
	}
	else = {
		set_variable = {
			name = sp_i32_bur_surplus_months
			value = 0
		}
	}
}

# CMF bar/hide effects require scope:journal_entry. Button/event paths may lack it.
sp_i32_ensure_journal_entry_scope = {
	if = {
		limit = {
			NOT = { exists = scope:journal_entry }
			has_journal_entry = sp_je_learning_from_europe
		}
		je:sp_je_learning_from_europe = {
			save_scope_as = journal_entry
		}
	}
}

sp_i32_apply_cmf_hide_flags = {
	sp_i32_ensure_journal_entry_scope = yes
	if = {
		limit = {
			exists = scope:journal_entry
		}
		scope:journal_entry = {
			set_variable = com_hide_modifiers
			set_variable = com_hide_scripted_buttons
			set_variable = com_hide_involved_countries
			set_variable = com_hide_complete_triggers
			set_variable = com_hide_fail_triggers
		}
	}
}

# Moltke presentation only — NO gameplay modifier.
# RUNTIME-PENDING: foreign PRU character on a TUR JE com_journal_characters list
# is unproven in CMF 1.64. status_desc remains the loc fallback if the portrait fails.
sp_i32_sync_moltke_character = {
	sp_i32_ensure_journal_entry_scope = yes
	if = {
		limit = {
			exists = scope:journal_entry
		}
		scope:journal_entry = {
			clear_variable_list = com_journal_characters
		}
		if = {
			limit = {
				sp_i32_has_incoming_military_assistance = yes
				game_date < 1841.1.1
			}
			if = {
				limit = {
					exists = c:PRU
				}
				c:PRU = {
					random_scope_character = {
						limit = {
							has_template = PRU_helmuth_von_moltke
						}
						save_scope_as = sp_i32_moltke
					}
				}
			}
			if = {
				limit = {
					NOT = { exists = scope:sp_i32_moltke }
					exists = c:GER
				}
				c:GER = {
					random_scope_character = {
						limit = {
							has_template = PRU_helmuth_von_moltke
						}
						save_scope_as = sp_i32_moltke
					}
				}
			}
			if = {
				limit = {
					exists = scope:sp_i32_moltke
				}
				scope:sp_i32_moltke = {
					set_variable = {
						name = com_opinion
						value = flag:sp_i32_opinion_moltke
					}
				}
				scope:journal_entry = {
					add_to_variable_list = {
						name = com_journal_characters
						target = scope:sp_i32_moltke
					}
				}
			}
		}
	}
}

		}
		unhide_com_progress_bar = { progress_bar = sp_i32_spb_education }
	}
	else = {
		hide_com_progress_bar = { progress_bar = sp_i32_spb_education }

# Applies the active Students modifier to the JE. Wave A destination options
# inline add_modifier instead, so the native tooltip is not collapsed.
sp_i32_start_students_shared = {
	set_variable = sp_i32_students_active
	if = {
		limit = {
			sp_i32_students_matches_primary = yes
		}
		set_variable = {
			name = sp_i32_students_duration_months
			value = 48
		}
		set_variable = {
			name = sp_i32_students_months_left
			value = 48
		}
	}
	else = {
		set_variable = {
			name = sp_i32_students_duration_months
			value = 60
		}
		set_variable = {
			name = sp_i32_students_months_left
			value = 60
		}
	}
	remove_variable = sp_i32_students_completing
	remove_variable = sp_i32_students_midpoint_fired
	remove_variable = sp_i32_students_chill
	remove_variable = sp_i32_students_forced_sink
	sp_i32_reset_students_bar = yes
	sp_i32_ensure_journal_entry_scope = yes
	if = {
		limit = {
			exists = scope:journal_entry
		}
		unhide_com_progress_bar = { progress_bar = sp_i32_spb_students }
	}
}

sp_i32_try_students_midpoint_complication = {
	if = {
		limit = {
			NOT = { has_variable = sp_i32_complication_lock }
			has_variable = sp_i32_students_active
			NOT = { has_variable = sp_i32_students_completing }
			NOT = { has_variable = sp_i32_students_midpoint_fired }
			has_variable = sp_i32_students_months_left
			has_variable = sp_i32_students_duration_months
			var:sp_i32_students_months_left <= {
				value = var:sp_i32_students_duration_months
				divide = 2
			}
		}
		set_variable = sp_i32_students_midpoint_fired
		if = {
			limit = {
				legitimacy < 40
			}
			random_list = {
				65 = { }
				35 = {
					set_variable = {
						name = sp_i32_complication_lock
						days = 30
					}
					trigger_event = { id = sp_i32.50 popup = yes }
				}
			}
		}
		else = {
			random_list = {
				80 = { }
				20 = {
					set_variable = {
						name = sp_i32_complication_lock
						days = 30
					}
					trigger_event = { id = sp_i32.50 popup = yes }
				}
			}
		}
	}
}

sp_i32_tick_students = {
	if = {
		limit = {
			has_variable = sp_i32_students_active
			NOT = { has_variable = sp_i32_students_completing }
		}
		if = {
			limit = {
				NOT = { sp_i32_students_partner_still_exists = yes }
			}
			set_variable = sp_i32_students_chill
		}
		change_variable = {
			name = sp_i32_students_months_left
			add = -1
		}
		if = {
			limit = {
				var:sp_i32_students_months_left <= 0
			}
			set_variable = sp_i32_students_completing
			trigger_event = { id = sp_i32.40 popup = yes }
		}
	}
}
