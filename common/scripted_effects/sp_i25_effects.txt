# I-25 yearly dispatch. At most one event per pulse. Cooldown is set in event immediate,
# not here: trigger_event may be queued, and setting cooldown first would fail the event trigger.

sp_i25_set_flavor_cooldown = {
	set_variable = { name = sp_i25_flavor_cooldown days = 547 }
}
