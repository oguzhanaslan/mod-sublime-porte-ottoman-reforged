### Ottomanism claim: none.

sp_i31_ottomanism_recount_progress = {
	set_variable = {
		name = sp_ottomanism_stage_progress
		value = 0
	}
	if = {
		limit = { var:sp_ottomanism_stage = 1 }
		if = {
			limit = { sp_i31_inclusive_citizenship = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { sp_i31_civic_church = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
	}
	else_if = {
		limit = { var:sp_ottomanism_stage = 2 }
		if = {
			limit = { sp_i31_constitutional_any_franchise = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { sp_i31_press_opening = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { government_legitimacy >= 50 }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
	}
	else_if = {
		limit = { var:sp_ottomanism_stage = 3 }
		if = {
			limit = { sp_i31_ottomanism_stage_3_complete = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
	}
	else_if = {
		limit = { var:sp_ottomanism_stage = 4 }
		if = {
			limit = { sp_i31_democratic_franchise = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { sp_i31_inclusive_citizenship = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { country_rank >= rank_value:great_power }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { literacy_rate >= 0.35 }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { sp_i31_modern_institutions = yes }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
		if = {
			limit = { government_legitimacy >= 60 }
			change_variable = { name = sp_ottomanism_stage_progress add = 1 }
		}
	}
}

sp_i31_clear_caliph_war_token_if_peace = {
	if = {
		limit = {
			has_variable = sp_caliph_calls_this_war
			is_at_war = no
		}
		remove_variable = sp_caliph_calls_this_war
	}
}

sp_i31_try_caliph_calls = {
	sp_i31_clear_caliph_war_token_if_peace = yes
	if = {
		limit = {
			has_variable = sp_islamism_selected
			has_journal_entry = sp_je_pan_islamic_unity
			var:sp_islamism_stage >= 3
			sp_i31_at_war_vs_christian_gp = yes
			NOT = { has_variable = sp_caliph_calls_cooldown }
			NOT = { has_variable = sp_caliph_calls_this_war }
			any_country = {
				sp_i31_eligible_caliph_call_target = yes
			}
		}
		set_variable = sp_caliph_calls_this_war
		set_variable = {
			name = sp_caliph_calls_cooldown
			months = 60
		}
		set_variable = {
			name = sp_caliph_call_batch
			value = 0
		}
		trigger_event = {
			id = sp_i31i.30
			popup = yes
		}
		while = {
			limit = {
				var:sp_caliph_call_batch < 5
				any_country = {
					sp_i31_eligible_caliph_call_target = yes
				}
			}
			ordered_country = {
				limit = {
					sp_i31_eligible_caliph_call_target = yes
				}
				order_by = gdp
				position = 0
				check_range_bounds = no
				set_variable = {
					name = sp_caliph_call_received
					months = 12
				}
				trigger_event = {
					id = sp_i31i.31
					popup = yes
				}
			}
			change_variable = {
				name = sp_caliph_call_batch
				add = 1
			}
		}
	}
}
