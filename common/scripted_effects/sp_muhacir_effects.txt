# State scope. Stamp only while this share is TUR-owned Rumelian candidate.
sp_i12_stamp_tur_share_provenance_on_this_state = {
	if = {
		limit = {
			sp_i12_can_stamp_tur_share_provenance = yes
			NOT = { has_variable = sp_i12_tur_share_provenance }
		}
		set_variable = sp_i12_tur_share_provenance
	}
}

# Country scope: TUR. Stamp every currently owned Rumelian Wave C candidate share.
# Intended caller: future additive on_actions (game start + state pulse). Not wired here.
sp_i12_stamp_all_owned_rumelian_share_provenance = {
	every_scope_state = {
		limit = {
			sp_i12_can_stamp_tur_share_provenance = yes
		}
		sp_i12_stamp_tur_share_provenance_on_this_state = yes
	}
}

# Country scope: TUR. Saves staging destination for Wave C moves.
sp_i12_save_wave_c_staging_state = {
	if = {
		limit = {
			s:STATE_EASTERN_THRACE = {
				any_scope_state = {
					owner ?= c:TUR
				}
			}
		}
		s:STATE_EASTERN_THRACE = {
			random_scope_state = {
				limit = {
					owner ?= c:TUR
				}
				save_scope_as = sp_i12_wave_c_staging_state
			}
		}
	}
	else = {
		random_scope_state = {
			limit = {
				is_capital = yes
			}
			save_scope_as = sp_i12_wave_c_staging_state
		}
	}
}

# State-region walk helper. Expects scope:sp_i12_wave_c_staging_state.
# Evacuates only provenance-stamped former TUR shares (B9).
# 1836 RUS Dobrudja / SER Eastern Serbia shares have no stamp -> skipped.
sp_i12_evacuate_foreign_shares_in_this_state_region = {
	every_scope_state = {
		limit = {
			sp_i12_is_lost_tur_rumelian_share = yes
		}
		every_scope_pop = {
			limit = {
				sp_i12_is_muhacir_refugee_pop = yes
			}
			move_partial_pop = {
				state = scope:sp_i12_wave_c_staging_state
				population_ratio = 0.75
			}
		}
	}
}

# Country scope: TUR. Still-owned devastated Rumelian shares.
# Own-state move_partial_pop + kill is attested (Hokkaido / balkan_wars.9 owner-side).
sp_i12_evacuate_devastated_held_rumelian_to_staging = {
	if = {
		limit = {
			NOT = {
				exists = scope:sp_i12_wave_c_staging_state
			}
		}
		sp_i12_save_wave_c_staging_state = yes
	}
	if = {
		limit = {
			exists = scope:sp_i12_wave_c_staging_state
		}
		every_scope_state = {
			limit = {
				sp_i12_is_devastated_held_rumelian_share = yes
				NOT = {
					this = scope:sp_i12_wave_c_staging_state
				}
			}
			every_scope_pop = {
				limit = {
					sp_i12_is_muhacir_refugee_pop = yes
				}
				move_partial_pop = {
					state = scope:sp_i12_wave_c_staging_state
					population_ratio = 0.75
				}
			}
			kill_population_percent_in_state = {
				percent = 0.10
				culture = cu:turkish
			}
			kill_population_percent_in_state = {
				percent = 0.10
				culture = cu:tatar
			}
			kill_population_percent_in_state = {
				percent = 0.10
				culture = cu:circassian
			}
		}
	}
}
