	transfer_state = { STATE = s:STATE_ALEPPO GIVER = c:EGY TAKER = c:TUR }

	transfer_state = { STATE = s:STATE_LEBANON GIVER = c:EGY TAKER = c:TUR }
	transfer_state = { STATE = s:STATE_PALESTINE GIVER = c:EGY TAKER = c:TUR }
