sp_i28_recount_stage_progress = {
	set_variable = {
		name = sp_turan_stage_progress
		value = 0
	}
	if = {
		limit = { var:sp_turan_stage = 1 }
		if = {
			limit = { country_or_subject_owns_entire_state_region = STATE_AZERBAIJAN }
			change_variable = { name = sp_turan_stage_progress add = 1 }
		}
		if = {
			limit = { country_or_subject_owns_entire_state_region = STATE_ELIZAVETPOL }
			change_variable = { name = sp_turan_stage_progress add = 1 }
		}
		if = {
			limit = { country_or_subject_owns_entire_state_region = STATE_TABRIZ }
			change_variable = { name = sp_turan_stage_progress add = 1 }
		}
		if = {
			limit = { country_or_subject_owns_entire_state_region = STATE_URMIA }
			change_variable = { name = sp_turan_stage_progress add = 1 }
		}
	}
	else_if = {
		limit = { var:sp_turan_stage = 2 }
		if = {
			limit = { country_or_subject_owns_entire_state_region = STATE_IRAKAJEMI }
			change_variable = { name = sp_turan_stage_progress add = 1 }
		}
		if = {
			limit = { country_or_subject_owns_entire_state_region = STATE_ISFAHAN }

sp_i28_try_add_turan_je = {
	if = {
		limit = {
			sp_i28_turan_eligible = yes
			NOT = { has_journal_entry = sp_je_turanian_ideal }
		}
		add_journal_entry = {
			type = sp_je_turanian_ideal
		}
	}
}

sp_i28_finalize_turan_reward = {
	set_variable = sp_turan_ideal_complete
	set_variable = {
		name = sp_turan_stage
		value = 6
	}
	add_modifier = {
		name = sp_unifier_of_turan
		months = -1
	}
}
