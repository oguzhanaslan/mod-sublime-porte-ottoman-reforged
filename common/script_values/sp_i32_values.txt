# I-32 Wave A-E - occupied families, capacity, and primary-partner pay twins. Not mana.

sp_i32_occupied_families = {
	value = 0
	if = {
		limit = {
			has_variable = sp_i32_students_active
		}
		add = 1
	}
	# Military family: MA OR I-32 land invite (never both - OR, not sum).
	if = {
		limit = {
			OR = {
				has_variable = sp_i32_military_active
				sp_i32_has_incoming_military_assistance = yes
			}
		}
		add = 1
	}
	if = {
		limit = {
			has_variable = sp_i32_technical_active
		}
		add = 1
	}
	if = {
		limit = {
			has_variable = sp_i32_education_active
		}
		add = 1
	}
	if = {
		limit = {
			has_variable = sp_i32_industrial_active
		}
		add = 1
	}
	if = {
		limit = {
			has_variable = sp_i32_naval_active
		}
		add = 1
	}
}

# 2 default; 3 if 12-month bureaucracy surplus or staffed university; 4 if Established.
sp_i32_capacity_max = {
	value = 2
	if = {
		limit = {
			OR = {
				AND = {
					has_variable = sp_i32_bur_surplus_months
					var:sp_i32_bur_surplus_months >= 12
				}
				sp_i32_has_staffed_university = yes
				sp_i32_expertise_established = yes
			}
		}
		add = 1
	}
	if = {
		limit = {
			sp_i32_expertise_established = yes
		}
		add = 1
	}
}

# Naked possible gold: cheapest diplomatic host including primary -15%.
# Uses diplomatic gates only (not *_eligible) to avoid circular gold checks.
sp_i32_students_gold_required = {
	if = {
		limit = {
			sp_i32_students_can_use_country = { PARTNER = c:AUS }
			has_variable = sp_i32_primary_partner_aus
		}
		value = sp_i32_students_upfront_austria_pri
	}
	else_if = {
		limit = {
			sp_i32_students_can_use_country = { PARTNER = c:AUS }
		}
		value = sp_i32_students_upfront_austria
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					sp_i32_students_can_use_country = { PARTNER = c:FRA }
					has_variable = sp_i32_primary_partner_fra
				}
				AND = {
					sp_i32_students_can_use_country = { PARTNER = c:GBR }
					has_variable = sp_i32_primary_partner_gbr
				}
				AND = {
					sp_i32_students_german_host_ok = yes
					has_variable = sp_i32_primary_partner_german
				}
			}
		}
		value = sp_i32_students_upfront_standard_pri
	}
	else = {
		value = sp_i32_students_upfront_standard
	}
}

# JE-button preview: Austria-only eligibility uses the cheaper cost; otherwise standard.
# Primary -15% when primary matches that cheapest path. Does not charge.
sp_i32_students_tooltip_pay = {
	if = {
		limit = {
			NOR = {
				sp_i32_students_fra_eligible = yes
				sp_i32_students_gbr_eligible = yes
				sp_i32_students_german_eligible = yes
			}
			sp_i32_students_aus_eligible = yes
			has_variable = sp_i32_primary_partner_aus
		}
		value = sp_i32_students_upfront_austria_pri
	}
	else_if = {
		limit = {
			NOR = {
				sp_i32_students_fra_eligible = yes
				sp_i32_students_gbr_eligible = yes
				sp_i32_students_german_eligible = yes
			}
			sp_i32_students_aus_eligible = yes
		}
		value = sp_i32_students_upfront_austria
	}
	else_if = {
		limit = {
			OR = {
				AND = {
					sp_i32_students_fra_eligible = yes
					has_variable = sp_i32_primary_partner_fra
				}
				AND = {
					sp_i32_students_gbr_eligible = yes
					has_variable = sp_i32_primary_partner_gbr
				}
				AND = {
					sp_i32_students_german_eligible = yes
					has_variable = sp_i32_primary_partner_german
				}
			}
		}
		value = sp_i32_students_upfront_standard_pri
	}
	else = {
		value = sp_i32_students_upfront_standard
	}
	multiply = -1
}

				}

					sp_i32_education_limit_diplomatic = yes

						has_variable = sp_i32_primary_partner_fra
						has_variable = sp_i32_primary_partner_german
