# I-32 Wave A — occupied families and capacity ceilings. Not mana.

sp_i32_occupied_families = {
	value = 0
	if = {
		limit = {
			has_variable = sp_i32_students_active
		}
		add = 1
	}
	if = {
		limit = {
			sp_i32_has_incoming_military_assistance = yes
		}
		add = 1
	}
}

# 2 default; 3 if 12-month bureaucracy surplus or staffed university; 4 if Established.
sp_i32_capacity_max = {
	value = 2
	if = {
		limit = {
			OR = {
				AND = {
					has_variable = sp_i32_bur_surplus_months
					var:sp_i32_bur_surplus_months >= 12
				}
				sp_i32_has_staffed_university = yes
				sp_i32_expertise_established = yes
			}
		}
		add = 1
	}
	if = {
		limit = {
			sp_i32_expertise_established = yes
		}
		add = 1
	}
}

# Single source of truth for Students upfront gold. Tooltips must use these
# via gold_reserves / add_treasury — never hardcode the figure in loc.
sp_i32_students_upfront_standard = {
	value = 8000
}

sp_i32_students_upfront_austria = {
	value = 6000
}

# Signed treasury deltas for native add_treasury tooltips.
sp_i32_students_pay_standard = {
	value = sp_i32_students_upfront_standard
	multiply = -1
}

sp_i32_students_pay_austria = {
	value = sp_i32_students_upfront_austria
	multiply = -1
}

# Naked possible gold: cheapest diplomatic host the player can actually pick.
# If Austria is a valid host, 6000 is enough to open the picker; otherwise 8000.
sp_i32_students_gold_required = {
	if = {
		limit = {
			sp_i32_students_can_use_country = { PARTNER = c:AUS }
		}
		value = sp_i32_students_upfront_austria
	}
	else = {
		value = sp_i32_students_upfront_standard
	}
}

# JE-button preview: Austria-only eligibility uses the cheaper cost; otherwise the standard cost.
# Does not charge. Authoritative add_treasury lives on sp_i32.11 options.
sp_i32_students_tooltip_pay = {
	if = {
		limit = {
			NOR = {
				sp_i32_students_fra_eligible = yes
				sp_i32_students_gbr_eligible = yes
				sp_i32_students_german_eligible = yes
			}
			sp_i32_students_aus_eligible = yes
		}
		value = sp_i32_students_upfront_austria
	}
	else = {
		value = sp_i32_students_upfront_standard
	}
	multiply = -1
}
