# I-31 umbrella JE — Ottomanism / The Ottoman Nation

sp_je_ottomanism = {
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	group = je_group_historical_content
	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes

	is_shown_when_inactive = {
		sp_i31_ottomanism_eligible = yes
	}

	possible = {
		sp_i31_ottomanism_eligible = yes
	}

	immediate = {
		set_variable = {
			name = sp_ottomanism_stage
			value = 1
		}
		set_variable = {
			name = sp_ottomanism_stage_goal
			value = 2
		}
		set_variable = {
			name = sp_ottomanism_stage_progress
			value = 0
		}
		sp_i31_ottomanism_recount_progress = yes
		trigger_event = {
			id = sp_i31o.2
			popup = yes
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = sp_ottomanism_stage_4_done
					NOT = { has_variable = sp_ottomanism_ideal_complete }
				}
				desc = sp_je_ottomanism_status_final
			}
			triggered_desc = {
				trigger = { var:sp_ottomanism_stage = 4 }
				desc = sp_je_ottomanism_status_stage_4
			}
			triggered_desc = {
				trigger = { var:sp_ottomanism_stage = 3 }
				desc = sp_je_ottomanism_status_stage_3
			}
			triggered_desc = {
				trigger = { var:sp_ottomanism_stage = 2 }
				desc = sp_je_ottomanism_status_stage_2
			}
			desc = sp_je_ottomanism_status_stage_1
		}
	}

	current_value = {
		value = var:sp_ottomanism_stage_progress
	}

	goal_add_value = {
		value = var:sp_ottomanism_stage_goal
	}

	progressbar = yes

	# Display = current-stage laws/rank/literacy (native). Pulse still uses
	# sp_i31_ottomanism_stage_*_complete. hidden_trigger blocks mid-path auto-complete.
	complete = {
		trigger_if = {
			limit = { var:sp_ottomanism_stage = 1 }
			OR = {
				has_law_or_variant = law_type:law_racial_segregation
				has_law_or_variant = law_type:law_cultural_exclusion
				has_law_or_variant = law_type:law_multicultural
			}
			NOR = {
				has_law_or_variant = law_type:law_ethnostate
				has_law_or_variant = law_type:law_national_supremacy
				has_law_or_variant = law_type:law_subjecthood
			}
			OR = {
				has_law_or_variant = law_type:law_freedom_of_conscience
				has_law_or_variant = law_type:law_total_separation
			}
		}
		trigger_if = {
			limit = { var:sp_ottomanism_stage = 2 }
			has_law = law_type:law_monarchy
			OR = {
				has_law_or_variant = law_type:law_landed_voting
				has_law_or_variant = law_type:law_wealth_voting
				has_law_or_variant = law_type:law_census_voting
				has_law_or_variant = law_type:law_universal_suffrage
			}
			OR = {
				has_law_or_variant = law_type:law_right_of_assembly
				has_law_or_variant = law_type:law_protected_speech
			}
			government_legitimacy >= 50
		}
		trigger_if = {
			limit = { var:sp_ottomanism_stage = 3 }
			OR = {
				has_law_or_variant = law_type:law_racial_segregation
				has_law_or_variant = law_type:law_cultural_exclusion
				has_law_or_variant = law_type:law_multicultural
			}
			NOR = {
				has_law_or_variant = law_type:law_ethnostate
				has_law_or_variant = law_type:law_national_supremacy
				has_law_or_variant = law_type:law_subjecthood
			}
			highest_secession_progress < 0.20
			NOT = {
				any_scope_state = {
					is_incorporated = yes
					turmoil >= 0.25
				}
			}
		}
		trigger_if = {
			limit = { var:sp_ottomanism_stage = 4 }
			has_law = law_type:law_monarchy
			OR = {
				has_law_or_variant = law_type:law_wealth_voting
				has_law_or_variant = law_type:law_census_voting
				has_law_or_variant = law_type:law_universal_suffrage
			}
			OR = {
				has_law_or_variant = law_type:law_racial_segregation
				has_law_or_variant = law_type:law_cultural_exclusion
				has_law_or_variant = law_type:law_multicultural
			}
			NOR = {
				has_law_or_variant = law_type:law_ethnostate
				has_law_or_variant = law_type:law_national_supremacy
				has_law_or_variant = law_type:law_subjecthood
			}
			country_rank >= rank_value:great_power
			literacy_rate >= 0.35
			OR = {
				institution_investment_level = {
					institution = institution_schools
					value >= 2
				}
				institution_investment_level = {
					institution = institution_police
					value >= 2
				}
				institution_investment_level = {
					institution = institution_home_affairs
					value >= 2
				}
			}
			government_legitimacy >= 60
		}
		hidden_trigger = {
			has_variable = sp_ottomanism_ideal_complete
		}
	}

	on_complete = {
		custom_tooltip = sp_je_ottomanism_complete_tt
	}

	invalid = {
		OR = {
			NOT = { c:TUR ?= this }
			AND = {
				NOT = { has_variable = sp_ottomanism_selected }
				NOT = { has_variable = sp_ottomanism_ideal_complete }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			sp_i31_ottomanism_recount_progress = yes

			if = {
				limit = {
					var:sp_ottomanism_stage = 1
					NOT = { has_variable = sp_ottomanism_stage_1_done }
					sp_i31_ottomanism_stage_1_complete = yes
				}
				trigger_event = {
					id = sp_i31o.10
					popup = yes
				}
			}
			else_if = {
				limit = {
					var:sp_ottomanism_stage = 2
					NOT = { has_variable = sp_ottomanism_stage_2_done }
					sp_i31_ottomanism_stage_2_complete = yes
				}
				trigger_event = {
					id = sp_i31o.11
					popup = yes
				}
			}
			else_if = {
				limit = {
					var:sp_ottomanism_stage = 3
					NOT = { has_variable = sp_ottomanism_stage_3_done }
					sp_i31_ottomanism_stage_3_complete = yes
				}
				trigger_event = {
					id = sp_i31o.12
					popup = yes
				}
			}
			else_if = {
				limit = {
					var:sp_ottomanism_stage = 4
					NOT = { has_variable = sp_ottomanism_stage_4_done }
					sp_i31_ottomanism_stage_4_complete = yes
				}
				trigger_event = {
					id = sp_i31o.20
					popup = yes
				}
			}
		}
	}
}
